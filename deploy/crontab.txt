# ==============================================================================
# EDUSFERA.BY — CRON SCHEDULE (CRONTAB)
# ==============================================================================
# Инструкция: замените /var/www/edusfera.by на реальный абсолютный путь к проекту на сервере.
# При необходимости укажите точный путь к PHP (например: /usr/bin/php8.2 или /usr/local/bin/php).
#
# ------------------------------------------------------------------------------
# РЕЖИМ 1: ЕДИНЫЙ ПЛАНИРОВЩИК LARAVEL (НАИБОЛЕЕ РЕКОМЕНДУЕМЫЙ ДЛЯ VPS/VDS)
# Запускается раз в минуту. Laravel сам управляет всеми расписаниями из bootstrap/app.php.
# ------------------------------------------------------------------------------
* * * * * cd /var/www/edusfera.by && php artisan schedule:run >> /dev/null 2>&1

# ------------------------------------------------------------------------------
# РЕЖИМ 2: ПОЭЛЕМЕНТНЫЙ CRONTAB (ДЛЯ ШАРИНГ-ХОСТИНГОВ cPanel / ISPmanager / Beget)
# Используйте этот блок, если хостинг не поддерживает единый schedule:run:
# ------------------------------------------------------------------------------

# 1. Освобождение просроченных бронирований уроков (каждую минуту)
* * * * * cd /var/www/edusfera.by && php artisan lessons:expire-locks >> /dev/null 2>&1

# 2. Публикация транзакционных исходящих событий / Outbox (каждую минуту)
* * * * * cd /var/www/edusfera.by && php artisan integration:publish-outbox >> /dev/null 2>&1

# 3. Автоматический перевод завершённых уроков в статус completed (каждые 10 минут)
*/10 * * * * cd /var/www/edusfera.by && php artisan lessons:complete >> /dev/null 2>&1

# 4. Сверка холдов балансов и платежей (каждый час в 00 минут)
0 * * * * cd /var/www/edusfera.by && php artisan wallet:reconcile-holds >> /dev/null 2>&1

# 5. Очистка старых пакетов очереди и завершённых задач (ежедневно в 00:00)
0 0 * * * cd /var/www/edusfera.by && php artisan queue:prune-batches --hours=48 >> /dev/null 2>&1
0 0 * * * cd /var/www/edusfera.by && php artisan queue:prune-failed --hours=168 >> /dev/null 2>&1

# 6. Биллинг: списание и продление подписок преподавателей (ежедневно в 02:00)
0 2 * * * cd /var/www/edusfera.by && php artisan edusfera:subscriptions-process-billing >> /dev/null 2>&1

# 7. Сверка уроков виртуального класса с ИИ (ежедневно в 03:00)
0 3 * * * cd /var/www/edusfera.by && php artisan reconcile:lessons-with-ai >> /dev/null 2>&1

# 8. Очистка старых записей outbox (ежедневно в 03:30)
30 3 * * * cd /var/www/edusfera.by && php artisan integration:cleanup-outbox >> /dev/null 2>&1

# 9. Системная диагностика работоспособности платформы (ежедневно в 06:00)
0 6 * * * cd /var/www/edusfera.by && php artisan health:check >> /dev/null 2>&1
